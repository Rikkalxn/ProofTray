cmake_minimum_required(VERSION 3.22)
project(ProofTray LANGUAGES CXX)

option(PROOFTRAY_BUILD_TESTS "Build portable core tests" ON)
if(PROOFTRAY_BUILD_TESTS AND NOT ANDROID)
    enable_testing()
endif()

add_subdirectory(core)

option(PROOFTRAY_BUILD_TIFF_TESTS "Build the streamed Log TIFF writer tests" OFF)
if(ANDROID OR PROOFTRAY_BUILD_TIFF_TESTS)
    enable_language(C)
    add_subdirectory(third_party/libtiff)
    add_library(prooftray_log_tiff STATIC core/src/log_tiff.cc)
    target_link_libraries(prooftray_log_tiff PUBLIC prooftray_core PRIVATE TIFF::tiff)
    set_target_properties(prooftray_log_tiff PROPERTIES CXX_EXTENSIONS OFF POSITION_INDEPENDENT_CODE ON)
    if(ANDROID)
        target_compile_options(prooftray_log_tiff PRIVATE -O2)
    endif()
    if(PROOFTRAY_BUILD_TIFF_TESTS AND NOT ANDROID)
        enable_testing()
        add_executable(prooftray_log_tiff_tests core/tests/log_tiff_test.cc)
        target_link_libraries(prooftray_log_tiff_tests PRIVATE prooftray_log_tiff TIFF::tiff)
        add_test(NAME log_tiff COMMAND prooftray_log_tiff_tests)
    endif()
endif()

option(PROOFTRAY_BUILD_NLOG_PROBE "Build the desktop RAW calibration and LUT reference tool" OFF)
if(ANDROID OR PROOFTRAY_BUILD_NLOG_PROBE)
    add_subdirectory(third_party/libraw)
endif()
if(PROOFTRAY_BUILD_NLOG_PROBE)
    add_executable(nlog_probe core/tools/nlog_probe.cc)
    target_link_libraries(nlog_probe PRIVATE prooftray_core prooftray_libraw)
    if(WIN32)
        set_target_properties(prooftray_libraw PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    endif()
endif()
if(ANDROID)
    if(PROOFTRAY_BUILD_HEIF)
        add_library(prooftray_heif SHARED app/src/main/cpp/heif_jni.cc)
        target_link_libraries(prooftray_heif PRIVATE prooftray_heif_core jnigraphics)
    endif()
    add_library(prooftray_raw SHARED app/src/main/cpp/raw_decode_jni.cc)
    target_link_libraries(prooftray_raw PRIVATE prooftray_libraw prooftray_core prooftray_log_tiff jnigraphics)
    add_library(prooftray SHARED app/src/main/cpp/catalog_jni.cc app/src/main/cpp/ptp_jni.cc app/src/main/cpp/color_lut_jni.cc)
    target_link_libraries(prooftray PRIVATE prooftray_core jnigraphics)
    set_target_properties(prooftray prooftray_raw PROPERTIES CXX_EXTENSIONS OFF)
endif()
