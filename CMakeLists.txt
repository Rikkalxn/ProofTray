cmake_minimum_required(VERSION 3.22)
project(ProofTray LANGUAGES CXX)

option(OPENBRIDGE_BUILD_TESTS "Build portable catalog tests" ON)
if(OPENBRIDGE_BUILD_TESTS AND NOT ANDROID)
    enable_testing()
endif()

add_subdirectory(core)

option(OPENBRIDGE_BUILD_NLOG_PROBE "Build the desktop RAW calibration and LUT reference tool" OFF)
if(ANDROID OR OPENBRIDGE_BUILD_NLOG_PROBE)
    add_subdirectory(third_party/libraw)
endif()
if(OPENBRIDGE_BUILD_NLOG_PROBE)
    add_executable(nlog_probe tools/nlog_probe.cc)
    target_include_directories(nlog_probe PRIVATE app/src/main/cpp)
    target_link_libraries(nlog_probe PRIVATE openbridge_core openbridge_libraw)
    if(WIN32)
        set_target_properties(openbridge_libraw PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_BINARY_DIR}")
    endif()
endif()
if(ANDROID)
    add_library(openbridge_raw SHARED app/src/main/cpp/raw_decode_jni.cc)
    target_link_libraries(openbridge_raw PRIVATE openbridge_libraw openbridge_core jnigraphics)
    target_compile_features(openbridge_raw PRIVATE cxx_std_20)
    set_target_properties(openbridge_raw PROPERTIES CXX_EXTENSIONS OFF)
    add_library(openbridge SHARED app/src/main/cpp/catalog_jni.cc app/src/main/cpp/ptp_jni.cc app/src/main/cpp/color_lut_jni.cc)
    target_link_libraries(openbridge PRIVATE openbridge_core jnigraphics)
    target_compile_features(openbridge PRIVATE cxx_std_20)
    set_target_properties(openbridge PROPERTIES CXX_EXTENSIONS OFF)
endif()
