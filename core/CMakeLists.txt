cmake_minimum_required(VERSION 3.22)
project(ProofTrayCore LANGUAGES CXX)

set(color_sources src/color_adjustments.cc src/color_histogram.cc src/color_lut.cc src/log_encoding.cc src/raw_rendering.cc)
add_library(prooftray_core STATIC src/catalog.cc src/ptp.cc src/nikon_pairing.cc src/nikon_location.cc
    ${color_sources} ../third_party/furble/Blowfish.cpp)
target_include_directories(prooftray_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(prooftray_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/furble")
target_compile_features(prooftray_core PUBLIC cxx_std_20)
set_target_properties(prooftray_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(ANDROID)
    set_source_files_properties(${color_sources} PROPERTIES COMPILE_OPTIONS "-O2")
endif()

if(ANDROID OR CMAKE_SYSTEM_NAME STREQUAL "iOS")
    set(heif_default ON)
else()
    set(heif_default OFF)
endif()
option(PROOFTRAY_BUILD_HEIF "Build the portable HEIF still-image decoder" ${heif_default})
if(PROOFTRAY_BUILD_HEIF)
    enable_language(C)
    add_subdirectory(../third_party/heif ${CMAKE_CURRENT_BINARY_DIR}/heif-codecs)
    add_library(prooftray_heif_core STATIC src/heif_image.cc)
    target_compile_features(prooftray_heif_core PUBLIC cxx_std_20)
    target_include_directories(prooftray_heif_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
    target_link_libraries(prooftray_heif_core PRIVATE prooftray_heif_codec)
    set_target_properties(prooftray_heif_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
    if(ANDROID)
        target_compile_options(prooftray_heif_core PRIVATE "$<$<CONFIG:Debug>:-O2>")
    endif()
endif()

option(PROOFTRAY_BUILD_TESTS "Build portable core tests" ON)
if(PROOFTRAY_BUILD_TESTS AND NOT ANDROID)
    enable_testing()
    function(add_core_test target name source)
        add_executable(${target} tests/${source})
        target_link_libraries(${target} PRIVATE prooftray_core)
        add_test(NAME ${name} COMMAND ${target})
    endfunction()
    add_core_test(prooftray_catalog_tests catalog catalog_test.cc)
    add_core_test(prooftray_ptp_tests ptp_codec ptp_test.cc)
    add_core_test(prooftray_color_tests color_lut color_lut_test.cc)
    add_core_test(prooftray_color_adjustments_tests color_adjustments color_adjustments_test.cc)
    add_core_test(prooftray_raw_rendering_tests raw_rendering raw_rendering_test.cc)
    add_core_test(prooftray_location_tests nikon_location nikon_location_test.cc)
endif()
