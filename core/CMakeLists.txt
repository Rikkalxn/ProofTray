cmake_minimum_required(VERSION 3.22)
project(ProofTrayCore LANGUAGES CXX)

set(color_sources src/color_adjustments.cc src/color_histogram.cc src/color_lut.cc src/log_encoding.cc src/raw_rendering.cc)
add_library(openbridge_core STATIC src/catalog.cc src/ptp.cc src/nikon_pairing.cc src/nikon_location.cc
    ${color_sources} ../third_party/furble/Blowfish.cpp)
target_include_directories(openbridge_core PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_include_directories(openbridge_core PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/../third_party/furble")
target_compile_features(openbridge_core PUBLIC cxx_std_20)
set_target_properties(openbridge_core PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
if(ANDROID)
    set_source_files_properties(${color_sources} PROPERTIES COMPILE_OPTIONS "-O2")
endif()

option(OPENBRIDGE_BUILD_TESTS "Build portable core tests" ON)
if(OPENBRIDGE_BUILD_TESTS AND NOT ANDROID)
    enable_testing()
    # The desktop target names are also used by existing CLion run configurations.
    function(add_core_test target name source)
        add_executable(${target} tests/${source})
        target_link_libraries(${target} PRIVATE openbridge_core)
        add_test(NAME ${name} COMMAND ${target})
    endfunction()
    add_core_test(openbridge_catalog_tests catalog catalog_test.cc)
    add_core_test(openbridge_ptp_tests ptp_codec ptp_test.cc)
    add_core_test(openbridge_color_tests color_lut color_lut_test.cc)
    add_core_test(openbridge_color_adjustments_tests color_adjustments color_adjustments_test.cc)
    add_core_test(openbridge_raw_rendering_tests raw_rendering raw_rendering_test.cc)
    add_core_test(openbridge_location_tests nikon_location nikon_location_test.cc)
endif()
